---
title: Retry behavior
description: Configure retry behavior for methods using `x-fern-retry` extension
---

Configure retry behavior for methods:

```protobuf title="retry_service.proto"
syntax = "proto3";

package userservice.v1;

service UserService {
  rpc ProcessPayment(ProcessPaymentRequest) returns (PaymentResult) {
    option (x_fern_retry) = '{
      "max_attempts": 3,
      "exponential_backoff": true,
      "initial_delay_ms": 1000,
      "max_delay_ms": 30000,
      "retry_on_status": ["UNAVAILABLE", "DEADLINE_EXCEEDED", "INTERNAL"]
    }';
  }
  
  rpc UploadFile(UploadFileRequest) returns (UploadFileResponse) {
    option (x_fern_retry) = '{
      "max_attempts": 5,
      "exponential_backoff": true,
      "initial_delay_ms": 2000,
      "max_delay_ms": 60000
    }';
  }
}
```
